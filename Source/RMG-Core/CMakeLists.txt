# Link n02 static library for Kaillera netplay support
# PRIVATE prevents n02 from being transitively linked into RMG.exe,
# which would create duplicate static variables (two copies of n02 state).
# --export-all-symbols forces the linker to export n02 functions from
# RMG-Core.dll even though CORE_EXPORT (__declspec(dllexport)) triggers
# selective export mode.
if (TARGET n02)
    target_compile_definitions(RMG-Core PRIVATE RMGK_HAVE_P2P_TRANSPORT)
    target_link_libraries(RMG-Core PRIVATE n02)
    target_include_directories(RMG-Core PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/../n02
    )
    if (WIN32)
        target_link_options(RMG-Core PRIVATE -Wl,--export-all-symbols)
    endif()
endif()

# FBNeo (headless lib core) — built by the fbneo ExternalProject in
# 3rdParty/CMakeLists.txt and linked in as a static archive. FBNEO_LIB/FBNEO_DIR
# are only defined when the ExternalProject was created.
if (TARGET fbneo AND DEFINED FBNEO_LIB)
    add_dependencies(RMG-Core fbneo)
    target_link_libraries(RMG-Core PRIVATE ${FBNEO_LIB})
    target_link_libraries(RMG-Core PRIVATE
        Threads::Threads
        $<$<PLATFORM_ID:Windows>:ws2_32>
        ${MINIZIP_LIBRARIES}
    )
    # FbneoRuntime.cpp only includes <burn.h>, which pulls in burnint.h ->
    # joyprocess.h (src/burn/devices) and driverlist.h. driverlist.h is
    # generated by gamelist.pl into the Meson build dir, so that must be on the
    # include path as well.
    target_include_directories(RMG-Core PRIVATE
        ${FBNEO_DIR}/src/burn
        ${FBNEO_DIR}/src/burn/devices
        ${FBNEO_DIR}/src/cpu
        ${FBNEO_DIR}/src/intf
        ${FBNEO_DIR}/src/intf/audio
        ${FBNEO_DIR}/src/burner/rmgk
        ${FBNEO_BUILD_DIR}
    )
endif()

target_include_directories(RMG-Core PRIVATE
